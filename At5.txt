select country from customer;

select first_name as nome, last_name as sobrenome, country as pais from customer
where country in ('Brazil', 'Canada', 'France')
order by last_name;

select * from track

select name as nome, milliseconds as tempo_milisegundos from track
where milliseconds between 240000 and 360000 order by milliseconds desc
